# ============================================================
# BackupAndUpdate (ROS6/ROS7 compatible + Server Policy Check)
#
# reboot-safe + reboot-loop protection:
#  - flow control via flags (:return inside :do{} only exits block)
#  - reboot commands are the very last statements
#  - persistent failure counters in BKPUPD-DAILY comment: "fw,os,osver"
#    after $maxTries consecutive failures the script STOPS initiating
#    reboots (manual reset: /system scheduler set BKPUPD-DAILY comment="")
# ============================================================

:local scriptName "BackupAndUpdate"
:local dailySchedulerName "BKPUPD-DAILY"
:local rebootSchedulerName "BKPUPD-REBOOT"
:local backupName "auto-backup"
:local backupPassword ""
:local backupPath ""
:local backupFullPath ""
:local flashAvailable false

# Server URL for policy check
:local policyServerName "http://SERVER_NAME"
:local CAUrl ($policyServerName . "/fullchain.pem")
:local CACertName "globalSignCA.crt"
:local policyCheckUrl ($policyServerName . "/routeros/force_update.php")

:local osUpdateNeeded false
:local firmwareUpdateNeeded false
:local routerboardAvailable false
:local currentVersion ""
:local latestVersion ""

# ----- flow control flags -----
:local readyToInstall false
:local readyToReboot  false
:local backupFailed   false

# ----- reboot-loop protection -----
:local maxTries 2
:local fwTries 0
:local osTries 0
:local osTriedVer ""
:local countersDirty false

# Detect RouterOS version
:global rosVersion [/system resource get version]
:global majorVersion [:pick $rosVersion 0 1]
:global isRos7 ($majorVersion = "7")

# Obtain the device model, serial number and firmware versions
:global rawModel [/system resource get board-name]
:global deviceSerial [/system routerboard get serial-number]
:global deviceName [/system identity get name]
:global currentFirmware [/system routerboard get current-firmware]
:global upgradeFirmware [/system routerboard get upgrade-firmware]

# Deterministic delay of 0–599s based on serial number
:local seed 0
:for i from=0 to=([:len $deviceSerial] - 1) do={
    :set seed (($seed * 31) + [:tonum [:pick $deviceSerial $i ($i + 1)]])
}
:local delaySec ($seed % 600)
:log info ("BackupAndUpdate: scheduled delay " . $delaySec . "s for serial " . $deviceSerial)
:delay ($delaySec . "s")

:log info ("BackupAndUpdate: === SCRIPT STARTED ===")

# ============================================================
# Step 1: Daily scheduler
# ============================================================

:do {
    :log info ("BackupAndUpdate: Step 1 start")

    :if ([:len [/system scheduler find name=$dailySchedulerName]] = 0) do={
        /system scheduler add \
            name=$dailySchedulerName \
            start-time=05:00:00 \
            interval=1d \
            on-event=$scriptName \
            policy=ftp,reboot,read,write,policy,test,password,sniff,sensitive,romon
        :log info ("BackupAndUpdate: daily scheduler created")
    } else={
        :log info ("BackupAndUpdate: daily scheduler exists")
    }

    :log info ("BackupAndUpdate: Step 1 OK")
} on-error={
    :log error ("BackupAndUpdate: Step 1 FAILED")
}

# ============================================================
# Step 1.5: Read persistent failure counters from scheduler comment
# ============================================================

:do {
    :local cmt [/system scheduler get $dailySchedulerName comment]
    :if ([:len $cmt] > 0) do={
        :local p1 [:find $cmt ","]
        :set fwTries [:tonum [:pick $cmt 0 $p1]]
        :local rest [:pick $cmt ($p1 + 1) [:len $cmt]]
        :local p2 [:find $rest ","]
        :set osTries [:tonum [:pick $rest 0 $p2]]
        :set osTriedVer [:pick $rest ($p2 + 1) [:len $rest]]
    }
    :log info ("BackupAndUpdate: counters fw=" . $fwTries . " os=" . $osTries . " osver=" . $osTriedVer)
} on-error={
    :set fwTries 0
    :set osTries 0
    :set osTriedVer ""
    :log info ("BackupAndUpdate: counters empty or malformed - starting from zero")
}

# ============================================================
# Step 2: Check RouterOS updates
# ============================================================

:do {
    :log info ("BackupAndUpdate: Step 2 start")

    :local rosVersion [/system resource get version]
    :local majorVersion [:pick $rosVersion 0 1]

    :if ($majorVersion = "7") do={
        :log info ("BackupAndUpdate: ROS 7 detected -> executing mode=http")
        :local cmd "/system package update set mode=http"
        :execute $cmd
        :delay 1s
    } else={
        :log info ("BackupAndUpdate: mode=http skipped (ROS6)")
    }

    /system package update check-for-updates
    :log info ("BackupAndUpdate: check started, waiting 15s")

    :delay 15s

    :set currentVersion [/system package update get installed-version]
    :log info ("BackupAndUpdate: current=" . $currentVersion)

    :set latestVersion [/system package update get latest-version]
    :log info ("BackupAndUpdate: latest=" . $latestVersion)

    :local updateStatus [/system package update get status]
    :log info ("BackupAndUpdate: status=" . $updateStatus)

    :if ($updateStatus = "New version is available") do={
        :set osUpdateNeeded true
        :log info ("BackupAndUpdate: update IS available")
    } else={
        :log info ("BackupAndUpdate: no update available")
    }

    # ----- OS counter maintenance -----
    :if ($osUpdateNeeded = true) do={
        # Target version changed since last attempts -> legitimate retry
        :if ($latestVersion != $osTriedVer) do={
            :if ($osTries > 0) do={
                :log info ("BackupAndUpdate: new target " . $latestVersion . " (was " . $osTriedVer . ") - resetting OS counter")
            }
            :set osTries 0
            :set osTriedVer ""
            :set countersDirty true
        }
    } else={
        :if ($osTries > 0) do={
            :log info ("BackupAndUpdate: OS is up to date - resetting OS counter")
            :set osTries 0
            :set osTriedVer ""
            :set countersDirty true
        }
    }

    :log info ("BackupAndUpdate: Step 2 OK")
} on-error={
    :log error ("BackupAndUpdate: Step 2 FAILED")
}

# ============================================================
# Step 3: Check RouterBOARD firmware
# ============================================================

:do {
    :log info ("BackupAndUpdate: Step 3 start")
    :log info ("BackupAndUpdate: current-fw=" . $currentFirmware)
    :log info ("BackupAndUpdate: upgrade-fw=" . $upgradeFirmware)

    :set routerboardAvailable true

    :if ($currentFirmware != $upgradeFirmware) do={
        :set firmwareUpdateNeeded true
        :log info ("BackupAndUpdate: firmware update needed")
    } else={
        :log info ("BackupAndUpdate: firmware up to date")
        # ----- firmware counter maintenance -----
        :if ($fwTries > 0) do={
            :log info ("BackupAndUpdate: firmware in sync - resetting FW counter")
            :set fwTries 0
            :set countersDirty true
        }
    }

    :log info ("BackupAndUpdate: Step 3 OK")
} on-error={
    :log error ("BackupAndUpdate: Step 3 FAILED")
}

# Persist counter resets (if any)
:if ($countersDirty = true) do={
    :do {
        /system scheduler set $dailySchedulerName comment=($fwTries . "," . $osTries . "," . $osTriedVer)
        :set countersDirty false
    } on-error={
        :log error ("BackupAndUpdate: failed to persist counters")
    }
}

# ============================================================
# Helper: fetch with HTTPS (version-aware)
# For ROS6: auto-imports CA certificate from server if missing
# ============================================================

:local fetchUrl do={
    :local url $1
    :local dst $2
    :local urlCA $3
    :local caName $4

    # Determine if HTTPS is used
    :local isHttps false
    :local httpMode http
    :if ([:pick $url 0 5] = "https") do={
        :set isHttps true
        :set httpMode https
    }

    :if ($isRos7 = true) do={
        /tool fetch url=$url mode=$httpMode check-certificate=yes dst-path=$dst
    } else={
        :if ($isHttps = true) do={
            # ROS6: check if CA certificate exists
            :local certExists false
            :foreach certId in=[/certificate find where name ~"^$caName" ] do={
                :set certExists true
            }
            :if ($certExists = false) do={
                :log info ("BKPUPD-UPDATER: CA certificate " . $caName . " not found - downloading from server " . $urlCA)
                :do {
                    # Remove old fullchain.pem if exists 
                    /file remove [ find name="fullchain.pem"]
                    # Download CA certificate (insecure, no cert yet)
                    /tool fetch url=$urlCA mode=https check-certificate=no dst-path=fullchain.pem
                    :delay 2s
                    # Verify file exists
                    :local fileId [/file find name="fullchain.pem"]
                    :if ([:len $fileId] = 0) do={
                        :log error "BKPUPD-UPDATER: failed to download CA certificate"
                        :error "CA download failed"
                    }
                    # Import certificate
                    /certificate import file-name=fullchain.pem passphrase="" name=$caName
                    :delay 1s
                    :log info ("BKPUPD-UPDATER: CA certificate " . $caName . " imported successfully")
                    # Cleanup temp file
                    /file remove [ find name="fullchain.pem"]
                } on-error={
                    :log error "BKPUPD-UPDATER: CA certificate setup FAILED"
                    /file remove [ find name="fullchain.pem"]
                    :error "CA setup failed"
                }
            }
        }

       :if ($isHttps = true) do={
            #disable crl
            /certificate settings set crl-download=no crl-use=no
            # Now fetch with certificate validation
            /tool fetch url=$url mode=https check-certificate=yes dst-path=$dst
            } else={
            # Now fetch with certificate validation
            /tool fetch url=$url mode=http dst-path=$dst
            }
    }
}

# ============================================================
# Helper: URL-encode a string (percent-encoding)
# Usage: :local encoded [$urlEncode "some string"]
# ============================================================
:local urlEncode do={
    :local input $1
    :local result ""
    :local enc ""
    :for i from=0 to=([:len $input] - 1) do={
        :local char [:pick $input $i ($i + 1)]
        :set enc $char
        :if ($char = "%") do={ :set enc "%25" }
        :if ($char = " ") do={ :set enc "%20" }
        :if ($char = "&") do={ :set enc "%26" }
        :if ($char = "+") do={ :set enc "%2B" }
        :if ($char = "#") do={ :set enc "%23" }
        :if ($char = "=") do={ :set enc "%3D" }
        :if ($char = "?") do={ :set enc "%3F" }
        :if ($char = ";") do={ :set enc "%3B" }
        :if ($char = "/") do={ :set enc "%2F" }
        :if ($char = ":") do={ :set enc "%3A" }
        :if ($char = "@") do={ :set enc "%40" }
        :if ($char = "'") do={ :set enc "%27" }
        :if ($char = "\"") do={ :set enc "%22" }
        :if ($char = "<") do={ :set enc "%3C" }
        :if ($char = ">") do={ :set enc "%3E" }
        :set result ($result . $enc)
    }
    :return $result
}

# ============================================================
# Step 3.5: ALWAYS report device state to server
#           (+ policy check if RouterOS update is pending)
# ============================================================

:do {
    :log info "BackupAndUpdate: Step 3.5 start - reporting state to server"

    :local safeModel [$urlEncode $rawModel]
    :local safeName  [$urlEncode $deviceName]

    :local reportVersion $latestVersion
    :if ([:len $reportVersion] = 0) do={ :set reportVersion $currentVersion }
    :if ([:len $reportVersion] = 0) do={ :set reportVersion "unknown" }

    :local checkUrl ($policyCheckUrl . "?new_version=" . $reportVersion . "&model=" . $safeModel . "&serial=" . $deviceSerial . "&name=" . $safeName . "&currentFirmware=" . $currentFirmware . "&upgradeFirmware=" . $upgradeFirmware)
    :local checkFile "force_update_check.txt"
    :local serverResponse "DISABLED"

    :foreach fileId in=[/file find name=$checkFile] do={
        /file remove $fileId
    }

    :log info ("BackupAndUpdate: fetching policy from: " . $checkUrl)
    $fetchUrl $checkUrl $checkFile $CAUrl $CACertName
    :delay 5s

    :local fileId [/file find name=$checkFile]
    :if ([:len $fileId] > 0) do={
        :set serverResponse [/file get $fileId contents]
        /file remove $fileId
    }

    :log info ("BackupAndUpdate: server response for v" . $reportVersion . ": " . $serverResponse)

    :if ($osUpdateNeeded = true) do={
        :if ($serverResponse != "ENABLED") do={
            :log warning ("BackupAndUpdate: update NOT authorized by server. Aborting RouterOS update.")
            :set osUpdateNeeded false
        } else={
            :log info ("BackupAndUpdate: update authorized by server.")
        }
    }

    :log info ("BackupAndUpdate: Step 3.5 OK")
} on-error={
    :log error ("BackupAndUpdate: Step 3.5 FAILED (server unreachable or fetch error)")
    :set osUpdateNeeded false
}

# ============================================================
# Step 4: RouterOS update PREPARATION (backup etc.)
#         Gated by OS failure counter to prevent reboot loops
# ============================================================

:if (($osUpdateNeeded = true) && ($osTries >= $maxTries)) do={
    :log error ("BackupAndUpdate: OS install failed " . $osTries . " consecutive times - SKIPPING (reboot loop protection). Manual reset: /system scheduler set " . $dailySchedulerName . " comment=\"\"")
}

:if (($osUpdateNeeded = true) && ($osTries < $maxTries)) do={
    :do {
        :log info ("BackupAndUpdate: Step 4 start - RouterOS update preparation")

        # Reboot scheduler FIRST - safety net in case script dies mid-way
        :if ([:len [/system scheduler find name=$rebootSchedulerName]] = 0) do={
            /system scheduler add \
                name=$rebootSchedulerName \
                start-time=startup \
                interval=0 \
                on-event=":delay 3m; /system script run BackupAndUpdate" \
                policy=ftp,reboot,read,write,policy,test,password,sniff,sensitive,romon
            :log info ("BackupAndUpdate: reboot scheduler created")
        }

        # Detect flash directory
        :do {
            :local test [/file get flash name]
            :set flashAvailable true
            :set backupPath "flash/"
            :log info ("BackupAndUpdate: flash directory is available")
        } on-error={
            :set flashAvailable false
            :set backupPath ""
            :log info ("BackupAndUpdate: flash directory not available, using root")
        }

        # Generate backup name from Serial Number, Version and Date
        :do {
            :local serial [/system routerboard get serial-number]

            :local dateRaw [/system clock get date]

            :local year ""
            :local monthNum ""
            :local day ""

            :if ([:pick $dateRaw 0 1] ~ "[0-9]") do={
                :set year     [:pick $dateRaw 0 4]
                :set monthNum [:pick $dateRaw 5 7]
                :set day      [:pick $dateRaw 8 10]
            } else={
                :set day  [:pick $dateRaw 4 6]
                :set year [:pick $dateRaw 7 11]

                :local monthStr [:pick $dateRaw 0 3]
                :set monthNum "01"
                :if ($monthStr = "jan") do={ :set monthNum "01" }
                :if ($monthStr = "feb") do={ :set monthNum "02" }
                :if ($monthStr = "mar") do={ :set monthNum "03" }
                :if ($monthStr = "apr") do={ :set monthNum "04" }
                :if ($monthStr = "may") do={ :set monthNum "05" }
                :if ($monthStr = "jun") do={ :set monthNum "06" }
                :if ($monthStr = "jul") do={ :set monthNum "07" }
                :if ($monthStr = "aug") do={ :set monthNum "08" }
                :if ($monthStr = "sep") do={ :set monthNum "09" }
                :if ($monthStr = "oct") do={ :set monthNum "10" }
                :if ($monthStr = "nov") do={ :set monthNum "11" }
                :if ($monthStr = "dec") do={ :set monthNum "12" }
            }

            :local safeVersion "unknown"
            :if ([:len $currentVersion] > 0) do={
                :set safeVersion $currentVersion
            }

            :set backupName ( "auto-backup-" . $serial . "-" . $safeVersion . "-" . $year . "-" . $monthNum . "-" . $day)
            :set backupFullPath ($backupPath . $backupName)
            :log info ("BackupAndUpdate: backup full name: " . $backupFullPath)
        } on-error={
            :log error ("BackupAndUpdate: failed to generate backup name")
            :set backupFailed true
        }

        :if ($backupFailed = false) do={
            # Remove old backups
            :local prefixLen ([:len $backupPath] + 12)
            :local prefix ($backupPath . "auto-backup-")

            :foreach fileId in=[/file find] do={
                :local fileName [/file get $fileId name]
                :local fileLen [:len $fileName]

                :if ($fileLen >= $prefixLen) do={
                    :if ([:pick $fileName 0 $prefixLen] = $prefix) do={
                        :if (([:pick $fileName ($fileLen - 7) $fileLen] = ".backup") || \
                             ([:pick $fileName ($fileLen - 4) $fileLen] = ".rsc")) do={
                            /file remove $fileId
                            :log info ("BackupAndUpdate: removed old file: " . $fileName)
                        }
                    }
                }
            }

            # Binary backup
            :log info ("BackupAndUpdate: creating binary backup")
            :if ($backupPassword = "") do={
                /system backup save dont-encrypt=yes name=$backupFullPath
            } else={
                /system backup save password=$backupPassword name=$backupFullPath
            }

            # Config export
            :export file=$backupFullPath

            # Wait for files
            :local backupReady false
            :local backupWait 0

            :while (($backupReady = false) && ($backupWait < 60)) do={
                :delay 1s
                :if (([:len [/file find name=($backupFullPath . ".backup")]] > 0) && \
                     ([:len [/file find name=($backupFullPath . ".rsc")]] > 0)) do={
                    :set backupReady true
                }
                :set backupWait ($backupWait + 1)
            }

            :if ($backupReady = false) do={
                :log error ("BackupAndUpdate: backup timeout after 60s - aborting update")
                :set backupFailed true
            } else={
                :log info ("BackupAndUpdate: backup files ready in " . $backupWait . "s")

                # Count this install attempt BEFORE issuing install
                :set osTries ($osTries + 1)
                :set osTriedVer $latestVersion
                :do {
                    /system scheduler set $dailySchedulerName comment=($fwTries . "," . $osTries . "," . $osTriedVer)
                } on-error={
                    :log error ("BackupAndUpdate: failed to persist OS counter")
                }

                :set readyToInstall true
            }
        }

    } on-error={
        :log error ("BackupAndUpdate: Step 4 FAILED")
        :set backupFailed true
        :set readyToInstall false
    }
}

# ============================================================
# Step 5: RouterBOARD firmware update (flash only, NO reboot here)
#         Gated by FW failure counter to prevent reboot loops
# ============================================================

:if (($firmwareUpdateNeeded = true) && ($fwTries >= $maxTries)) do={
    :log error ("BackupAndUpdate: firmware upgrade failed " . $fwTries . " consecutive times - SKIPPING (reboot loop protection). Manual reset: /system scheduler set " . $dailySchedulerName . " comment=\"\"")
}

:if (($firmwareUpdateNeeded = true) && ($readyToInstall = false) && ($backupFailed = false) && ($fwTries < $maxTries)) do={
    :do {
        :log info ("BackupAndUpdate: Step 5 start - firmware update")

        :foreach schedulerId in=[/system scheduler find name=$rebootSchedulerName] do={
            /system scheduler remove $schedulerId
        }

        /system scheduler add \
            name=$rebootSchedulerName \
            start-time=startup \
            interval=0 \
            on-event=":delay 3m; /system script run BackupAndUpdate" \
            policy=ftp,reboot,read,write,policy,test,password,sniff,sensitive,romon

        # Count this attempt BEFORE flashing
        :set fwTries ($fwTries + 1)
        :do {
            /system scheduler set $dailySchedulerName comment=($fwTries . "," . $osTries . "," . $osTriedVer)
        } on-error={
            :log error ("BackupAndUpdate: failed to persist FW counter")
        }

        :log info ("BackupAndUpdate: upgrading firmware " . $currentFirmware . " -> " . $upgradeFirmware . " (attempt " . $fwTries . "/" . $maxTries . ")")
        /system routerboard upgrade

        :delay 5s

        :set readyToReboot true

    } on-error={
        :log error ("BackupAndUpdate: Step 5 FAILED")
        :set readyToReboot false
    }
}

# ============================================================
# Step 6: Cleanup - ONLY when no reboot is pending.
#         Reboot scheduler is removed ONLY if firmware is in sync.
# ============================================================

:if (($readyToInstall = false) && ($readyToReboot = false)) do={
    :log info ("BackupAndUpdate: Step 6 - cleanup")

    :if ($currentFirmware = $upgradeFirmware) do={
        :foreach schedulerId in=[/system scheduler find name=$rebootSchedulerName] do={
            /system scheduler remove $schedulerId
        }
        :log info ("BackupAndUpdate: firmware in sync - reboot scheduler removed")
    } else={
        :log warning ("BackupAndUpdate: firmware NOT in sync (" . $currentFirmware . " != " . $upgradeFirmware . ") - keeping reboot scheduler")
    }
} else={
    :log info ("BackupAndUpdate: Step 6 skipped - reboot pending, cleanup untouched")
}

:log info ("BackupAndUpdate: === SCRIPT FINISHED ===")

# ============================================================
# FINAL: reboot-inducing commands are the VERY LAST statements.
# ============================================================

:if ($readyToInstall = true) do={
    :log info ("BackupAndUpdate: installing RouterOS " . $latestVersion . " (attempt " . $osTries . "/" . $maxTries . ")")
    :log warning ("BackupAndUpdate: REBOOTING NOW (OS install)")
    /system package update install
}

:if ($readyToReboot = true) do={
    :log warning ("BackupAndUpdate: REBOOTING NOW (firmware activation, attempt " . $fwTries . "/" . $maxTries . ")")
    /system reboot
}
