# ============================================================
# BackupAndUpdate Updater (ROS6/ROS7 compatible, HTTP)
# ============================================================

:local scriptName "BackupAndUpdate"
:local scriptNameOld "BackupAndUpdate.old"
:local MyName "BKPUPD-UPDATER"
:local serverName "http://SERVER_NAME"
:local scriptBase ($serverName . "/routeros")
:local md5Url ($scriptBase . "/BackupAndUpdate.md5")
:local scriptUrl ($scriptBase . "/BackupAndUpdate.rsc")
:local tempMd5File "updater-md5.txt"
:local tempScriptFile "updater-script.rsc"

:log info ("BKPUPD-UPDATER: === UPDATER STARTED ===")

# ============================================================
# Step 1: Detect RouterOS version
# ============================================================

:local rosVersion [/system resource get version]
:local majorVersion [:pick $rosVersion 0 1]
:global isRos7 ($majorVersion = "7")

# ============================================================
# Helper: fetch with HTTP
# ============================================================

:local fetchUrl do={
    :local url $1
    :local dst $2
    /tool fetch url=$url mode=http dst-path=$dst
}

# ============================================================
# Step 2: Download MD5 hash from server
# ============================================================

:local serverMd5 ""

:do {
    :log info ("BKPUPD-UPDATER: fetching MD5 from " . $md5Url)

    :foreach fileId in=[/file find name=$tempMd5File] do={
        /file remove $fileId
    }

    $fetchUrl $md5Url $tempMd5File
    :delay 2s

    :local fileId [/file find name=$tempMd5File]
    :if ([:len $fileId] > 0) do={
        :set serverMd5 [/file get $fileId contents]
        :set serverMd5 [:pick $serverMd5 0 32]
        /file remove $fileId
        :log info ("BKPUPD-UPDATER: server MD5 = " . $serverMd5)
    } else={
        :log error "BKPUPD-UPDATER: failed to read MD5 file"
    }
} on-error={
    :log error "BKPUPD-UPDATER: failed to download MD5 (server unreachable or HTTP error?)"
}

:if ($serverMd5 = "") do={
    :log error "BKPUPD-UPDATER: no valid MD5 received - aborting"
    :error "No MD5"
}

# ============================================================
# Step 3: Read current MD5 from the main script's comment
# ============================================================

:local currentMd5 ""

:do {
    :set currentMd5 [/system script get $MyName comment]
    :if ([:len $currentMd5] > 0) do={
        :log info ("BKPUPD-UPDATER: current MD5 = " . $currentMd5)
    } else={
        :log info "BKPUPD-UPDATER: no current MD5 in script comment (first run?)"
    }
} on-error={
    :log warning "BKPUPD-UPDATER: failed to read script comment"
}

# ============================================================
# Step 4: Compare hashes and update if needed
# ============================================================

:if ($serverMd5 != $currentMd5) do={
    :log info "BKPUPD-UPDATER: MD5 mismatch - updating script"
    :do {
        :log info ("BKPUPD-UPDATER: Rename current script to: ".$scriptNameOld)
        /system script set [find name=$scriptName] name=$scriptNameOld
        }

    :do {
        :foreach fileId in=[/file find name=$tempScriptFile] do={
            /file remove $fileId
        }

        :log info ("BKPUPD-UPDATER: downloading script from " . $scriptUrl)
        $fetchUrl $scriptUrl $tempScriptFile
        :delay 2s

        :local fileId [/file find name=$tempScriptFile]
        :if ([:len $fileId] = 0) do={
            :log error "BKPUPD-UPDATER: failed to download script file"
            :error "Download failed"
        }

        :log info "BKPUPD-UPDATER: importing new script"
        /import file=$tempScriptFile
        :delay 1s

        # Persist new MD5 in the main script's comment
        /system script set $MyName comment=$serverMd5
        :log info ("BKPUPD-UPDATER: script updated successfully, new MD5 = " . $serverMd5)

        # Cleanup temp file
        :foreach fileId in=[/file find name=$tempScriptFile] do={
            /file remove $fileId
        }
    :do {
        :log info ("BKPUPD-UPDATER: Remove old script ".$scriptNameOld)
        /system script remove [ find name=$scriptNameOld ]
        }
    } on-error={
        :log error "BKPUPD-UPDATER: script update FAILED"
        :foreach fileId in=[/file find name=$tempScriptFile] do={
            /file remove $fileId
        }
       :do {
           :log info ("BKPUPD-UPDATER: Restore old script ".$scriptName)
           /system script set [find name=$scriptNameOld] name=$scriptName
           }
    }
} else={
    :log info "BKPUPD-UPDATER: script is up to date"
}

:log info "BKPUPD-UPDATER: === UPDATER FINISHED ==="
